cmake_minimum_required(VERSION 3.20)
project(capnp_codegen_oracle LANGUAGES C CXX)
find_package(CapnProto 1.2 CONFIG REQUIRED)
set(CAPNPC_SRC_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../Schema")
capnp_generate_cpp(sources headers ../Schema/common.capnp ../Schema/probe.capnp)
add_executable(capnp-codegen-oracle oracle.cpp ${sources} ${headers})
target_include_directories(capnp-codegen-oracle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(capnp-codegen-oracle PRIVATE CapnProto::capnp)
target_compile_features(capnp-codegen-oracle PRIVATE cxx_std_17)
