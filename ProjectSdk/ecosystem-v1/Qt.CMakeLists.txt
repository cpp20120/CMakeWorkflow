cmake_minimum_required(VERSION 3.26)
project(@NAME@ VERSION 0.1.0 LANGUAGES CXX)
include(cmake/cmwf-sdk/ecosystem-v1/Native.cmake)
include(cmake/cmwf-config/v1/Variant.cmake)
include(cmake/cmwf-sdk/ecosystem-v1/Qt.cmake)
option(BUILD_TESTING "Build Qt tests." ON)
cmwf_qt_setup()

add_library(@NAME@_ui STATIC src/window.cpp src/window.hpp src/window.ui)
target_include_directories(@NAME@_ui PUBLIC src)
target_link_libraries(@NAME@_ui PUBLIC Qt6::Widgets)
cmwf_sdk_target(@NAME@_ui PRIVATE)
cmwf_variant_apply(@NAME@_ui)
qt_add_executable(@NAME@ src/main.cpp src/app.qrc)
target_link_libraries(@NAME@ PRIVATE @NAME@_ui)
cmwf_sdk_target(@NAME@ PRIVATE)
cmwf_variant_apply(@NAME@)
if(BUILD_TESTING)
    enable_testing()
    qt_add_executable(@NAME@_test tests/smoke.cpp src/app.qrc)
    target_link_libraries(@NAME@_test PRIVATE @NAME@_ui Qt6::Test)
    cmwf_sdk_target(@NAME@_test PRIVATE)
    cmwf_variant_apply(@NAME@_test)
    add_test(NAME @NAME@.qt-test COMMAND @NAME@_test)
    add_test(NAME @NAME@.event-loop COMMAND @NAME@ --smoke)
    set_tests_properties(@NAME@.qt-test @NAME@.event-loop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
endif()
