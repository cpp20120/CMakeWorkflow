cmake_minimum_required(VERSION 3.26)
project(@NAME@ VERSION 0.1.0 LANGUAGES CXX)
include(cmake/cmwf-sdk/ecosystem-v1/Native.cmake)
include(cmake/cmwf-config/v1/Variant.cmake)
include(cmake/cmwf-sdk/ecosystem-v1/Graphics.cmake)
option(BUILD_TESTING "Build shader smoke tests." ON)
option(CMWF_VULKAN_RUNTIME_TEST "Create a Vulkan instance with an installed ICD." OFF)

add_executable(@NAME@ src/main.cpp)
target_link_libraries(@NAME@ PRIVATE Vulkan::Vulkan)
cmwf_sdk_target(@NAME@ PRIVATE)
cmwf_variant_apply(@NAME@)
cmwf_sdk_shader(@NAME@ "${PROJECT_SOURCE_DIR}/shaders/probe.comp" shader)
if(BUILD_TESTING)
    enable_testing()
    add_test(NAME @NAME@.spirv COMMAND @NAME@ "${shader}")
    if(CMWF_VULKAN_RUNTIME_TEST)
        add_test(NAME @NAME@.vulkan COMMAND @NAME@ --vulkan)
    endif()
endif()
