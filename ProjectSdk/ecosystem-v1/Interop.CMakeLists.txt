cmake_minimum_required(VERSION 3.26)
project(@NAME@ VERSION 0.1.0 LANGUAGES C CXX)
include(cmake/cmwf-sdk/ecosystem-v1/Native.cmake)
include(cmake/cmwf-config/v1/Variant.cmake)
include(cmake/cmwf-sdk/ecosystem-v1/Interop.cmake)
option(BUILD_TESTING "Build ABI tests." ON)

add_library(@NAME@ ${CMWF_VARIANT_LIBRARY_TYPE} src/interop.cpp)
target_sources(@NAME@ PUBLIC FILE_SET HEADERS BASE_DIRS include FILES include/@NAME@/api.h)
cmwf_interop_library(@NAME@)
if(BUILD_TESTING)
    enable_testing()
    add_executable(@NAME@_smoke tests/smoke.c)
    target_link_libraries(@NAME@_smoke PRIVATE @NAME@::@NAME@)
    cmwf_variant_apply(@NAME@_smoke)
    add_test(NAME @NAME@.c COMMAND @NAME@_smoke)
    add_executable(@NAME@_abi tests/abi.cpp)
    target_link_libraries(@NAME@_abi PRIVATE @NAME@::@NAME@)
    cmwf_variant_apply(@NAME@_abi)
    add_test(NAME @NAME@.abi COMMAND @NAME@_abi)
    cmwf_interop_managed_test(@NAME@)
endif()
include(CPack)
