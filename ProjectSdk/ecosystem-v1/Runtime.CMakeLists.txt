cmake_minimum_required(VERSION 3.26)
project(@NAME@ VERSION 0.1.0 LANGUAGES CXX)
include(cmake/cmwf-sdk/ecosystem-v1/Native.cmake)
include(cmake/cmwf-config/v1/Variant.cmake)
include(cmake/cmwf-sdk/ecosystem-v1/Runtime.cmake)
option(BUILD_TESTING "Build runtime tests." ON)

add_library(@NAME@ ${CMWF_VARIANT_LIBRARY_TYPE} src/queue.cpp)
target_sources(@NAME@ PUBLIC FILE_SET HEADERS BASE_DIRS include FILES include/@NAME@/queue.hpp)
cmwf_runtime_library(@NAME@)
add_executable(@NAME@_bench bench/benchmark.cpp)
target_link_libraries(@NAME@_bench PRIVATE @NAME@::@NAME@)
cmwf_sdk_target(@NAME@_bench PRIVATE)
cmwf_variant_apply(@NAME@_bench)
if(BUILD_TESTING)
    enable_testing()
    add_executable(@NAME@_smoke tests/smoke.cpp)
    target_link_libraries(@NAME@_smoke PRIVATE @NAME@::@NAME@)
    cmwf_sdk_target(@NAME@_smoke PRIVATE)
    cmwf_variant_apply(@NAME@_smoke)
    add_test(NAME @NAME@.queue COMMAND @NAME@_smoke)
endif()
if(NOT CMWF_VARIANT_INSTRUMENTED)
    include(CPack)
endif()
