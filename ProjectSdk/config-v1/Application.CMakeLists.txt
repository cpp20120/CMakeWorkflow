cmake_minimum_required(VERSION 3.26)
project(@NAME@ VERSION 0.1.0 LANGUAGES CXX)
include(cmake/cmwf-sdk/v1/Native.cmake)
include(cmake/cmwf-config/v1/Variant.cmake)
option(BUILD_TESTING "Build the smoke test." ON)

add_executable(@NAME@ src/main.cpp)
if(CMWF_VARIANT_INSTRUMENTED)
    cmwf_sdk_target(@NAME@ PRIVATE)
else()
    cmwf_sdk_application(@NAME@)
    include(CPack)
endif()
cmwf_variant_apply(@NAME@)
if(BUILD_TESTING)
    enable_testing()
    add_test(NAME @NAME@.smoke COMMAND @NAME@)
endif()
