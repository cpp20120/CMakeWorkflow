cmake_minimum_required(VERSION 3.20)
project(cmwf_capnp_oracle LANGUAGES C CXX)
find_package(CapnProto 1.2 CONFIG REQUIRED)
set(CAPNPC_SRC_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../Schema")
capnp_generate_cpp(workflow_sources workflow_headers ../Schema/workflow.capnp ../Schema/productionWorkflow.capnp)
add_executable(cmwf-capnp-oracle oracle.cpp ${workflow_sources} ${workflow_headers})
target_include_directories(cmwf-capnp-oracle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(cmwf-capnp-oracle PRIVATE CapnProto::capnp)
target_compile_features(cmwf-capnp-oracle PRIVATE cxx_std_17)
